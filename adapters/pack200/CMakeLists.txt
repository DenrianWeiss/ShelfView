set(pack_source "${PROJECT_SOURCE_DIR}/third_party/pack200/src/jdk.pack/share/native/common-unpack")
set(pack_generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${pack_generated}")
set(pack_files)
foreach(name bands bytes coding unpack utils zip)
  file(READ "${pack_source}/${name}.cpp" content)
  string(REGEX REPLACE "([^A-Za-z0-9_])::free\\(" "\\1bc_pack_free(" content "${content}")
  string(REGEX REPLACE "([^A-Za-z0-9_])::realloc\\(" "\\1bc_pack_realloc(" content "${content}")
  string(REPLACE "malloc(msize)" "bc_pack_malloc(msize)" content "${content}")
  file(WRITE "${pack_generated}/${name}.cpp" "#include \"pack_memory.h\"\n${content}")
  list(APPEND pack_files "${pack_generated}/${name}.cpp")
endforeach()
add_library(bcontrol_pack200 STATIC extract.cpp ${pack_files})
target_include_directories(bcontrol_pack200 PRIVATE "${pack_source}" "${CMAKE_CURRENT_SOURCE_DIR}" "${PROJECT_SOURCE_DIR}/adapters" "${PROJECT_SOURCE_DIR}/include")
target_compile_definitions(bcontrol_pack200 PRIVATE PRODUCT UNPACK_JNI)
include(TestBigEndian)
test_big_endian(pack_big_endian)
if(pack_big_endian)
  target_compile_definitions(bcontrol_pack200 PRIVATE _BIG_ENDIAN)
endif()
target_link_libraries(bcontrol_pack200 PRIVATE ZLIB::ZLIB)
set_target_properties(bcontrol_pack200 PROPERTIES CXX_STANDARD 11 POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)

if(NOT MSVC)
  target_compile_options(bcontrol_pack200 PRIVATE -Wno-error -Wno-deprecated-declarations)
endif()
